const Anchor = styled.a({
  // base styles

  [style.attribute('href').startsWith(
    'http',
    'https'
  )]: {
    // styles ...
  },

  // you need to use Styled's combinator helper `or` to create
  // nested selector lists (ie: comma separated selector)
  [style.or(
    '[href^="http"]',
    '[href^="https"]'
  )]: {
    // styles ...
  },

  // or have to duplicate style definitions
  '[href^="http"]': { /*  styles ... */ },
  '[href^="https"]': { /*  styles ... */ }
});

const Composition = () => (
  <Anchor href='#'>
    I'm an anchor
  </Anchor>
);

Refer to API section for available methods on attribute builder.
