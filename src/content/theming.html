CSS variable themes can be used as part of global style definitions.

const AppWrapper = styled
  .div()
  .withCSS(
    `:root {
      --background-primary: #fff;
      --color-primary: #262223;
    }`,

    `@media (prefers-color-scheme: dark) {
      :root {
        --background-primary: #121212;
        --color-primary: #fff;
      } 
    }`
  );
